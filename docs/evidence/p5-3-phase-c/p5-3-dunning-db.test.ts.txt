// 1. DATABASE_URL: Host localhost/127.0.0.1, Port 5432, Datenbank genau twir_g6_test; keine Verbindungs-Overrides per Query.
// 2. P53_CONFIRM_DB muss genau 'twir_g6_test' sein.   3. 'business_webapp' wird überall ausdrücklich ausgeschlossen.

const FORBIDDEN_DB = /business_webapp/i
const FORBIDDEN_QUERY_PARAMS = ['host', 'hostaddr', 'port', 'dbname', 'service', 'options']
function guardEnv() {
  const raw = process.env.DATABASE_URL
  if (!raw) throw new Error('ABBRUCH: DATABASE_URL ist nicht gesetzt.')
  if (FORBIDDEN_DB.test(raw)) throw new Error('ABBRUCH: DATABASE_URL verweist auf business_webapp (ausdrücklich verboten).')
  if (FORBIDDEN_DB.test(process.env.P53_CONFIRM_DB ?? '')) throw new Error('ABBRUCH: P53_CONFIRM_DB verweist auf business_webapp.')
  if (process.env.P53_CONFIRM_DB !== EXPECT_DB) throw new Error(`ABBRUCH: P53_CONFIRM_DB muss '${EXPECT_DB}' sein.`)
  let u: URL
  try { u = new URL(raw) } catch { throw new Error('ABBRUCH: DATABASE_URL ist keine gültige URL.') }
  if (u.protocol !== 'postgresql:' && u.protocol !== 'postgres:') throw new Error(`ABBRUCH: Protokoll '${u.protocol}' nicht erlaubt.`)
  const db = decodeURIComponent(u.pathname.replace(/^\//, ''))
  if (!['localhost', '127.0.0.1'].includes(u.hostname)) throw new Error(`ABBRUCH: Host '${u.hostname}' nicht erlaubt.`)
  if ((u.port || '5432') !== '5432') throw new Error(`ABBRUCH: Port '${u.port}' nicht erwartet (5432).`)
  if (db !== EXPECT_DB) throw new Error(`ABBRUCH: Datenbank '${db}' ist nicht '${EXPECT_DB}'.`)
  for (const key of u.searchParams.keys()) {
    if (FORBIDDEN_QUERY_PARAMS.includes(key.toLowerCase())) throw new Error(`ABBRUCH: URL-Parameter '${key}' ist nicht erlaubt (könnte Host/Datenbank überschreiben).`)
  }
}

async function inv(id: string) {
  const r = await prisma.$queryRaw`SELECT status::text AS status, type::text AS type, invoice_number, total_gross::text AS gross,
    paid_amount::text AS paid, due_date, sent_at FROM invoices WHERE id = ${id}`
  return r[0] ?? null
}
async function notices(id: string) {
  return prisma.$queryRaw`SELECT id, level, level_label, sent_at, due_date, fee::text AS fee, created_by_id, deleted_at

}

async function dispatchEvidence(id: string) {
  const r = await prisma.$queryRaw`SELECT a.id AS audit_id, a.created_at AS audit_created_at, i.sent_at AS sent_at
    FROM audit_logs a JOIN invoices i ON i.id = a.entity_id
    WHERE a.entity_id = ${id} AND a.entity_type = 'invoice' AND a.action = 'STATUS_CHANGE'::"AuditAction"
      AND a.user_id IS NOT NULL AND a.old_value->>'status' = 'FINALIZED' AND a.new_value->>'status' = 'SENT'
    ORDER BY a.created_at ASC`
  return r as Array<{ audit_id: string; audit_created_at: Date; sent_at: Date }>
}

const OWN_LIKE = `${TAG} ${RUN}%`
async function assertOwn(id: string, what: string) {
  if (typeof id !== 'string' || !id) throw new Error(`OWN-GUARD: ${what}: ungültige ID`)
  if (!created.includes(id)) throw new Error(`OWN-GUARD: ${what}: ${id} ist nicht in der Own-ID-Menge`)
  const r = await prisma.$queryRaw`SELECT i.id FROM invoices i WHERE i.id = ${id} AND (
      i.intro_text LIKE ${OWN_LIKE}
      OR (i.original_invoice_id = ANY(${created}::text[]) AND EXISTS (SELECT 1 FROM invoices o WHERE o.id = i.original_invoice_id AND o.intro_text LIKE ${OWN_LIKE})))`
  if (r.length !== 1) throw new Error(`OWN-GUARD: ${what}: ${id} trägt nicht das P5-3-Präfix dieses Laufs – KEIN Schreibzugriff`)
}
type OwnPatch =
  | { op: 'dueDate'; value: Date | null }
  | { op: 'sentNow' }                                   // status = SENT, sent_at = now()
  | { op: 'sentAt'; value: Date | null }
  | { op: 'sentAtFromAudit'; auditId: string; deltaMs: number }   // sent_at = Audit.created_at − deltaMs, komplett in PostgreSQL (keine JS-Date-/Zeitzonenumrechnung)
  | { op: 'type'; value: 'CORRECTION' }
  | { op: 'status'; value: 'OVERDUE' }
  | { op: 'paidAmount'; value: string }
async function ownSet(id: string, patch: OwnPatch) {
  await assertOwn(id, `ownSet(${patch.op})`)
  let n = -1
  switch (patch.op) {
    case 'dueDate':    n = await prisma.$executeRaw`UPDATE invoices SET due_date = ${patch.value} WHERE id = ${id} AND id = ANY(${created}::text[]) AND (intro_text LIKE ${OWN_LIKE} OR (original_invoice_id = ANY(${created}::text[]) AND EXISTS (SELECT 1 FROM invoices o WHERE o.id = invoices.original_invoice_id AND o.intro_text LIKE ${OWN_LIKE})))`; break
    case 'sentNow':    n = await prisma.$executeRaw`UPDATE invoices SET status = 'SENT'::"InvoiceStatus", sent_at = now() WHERE id = ${id} AND id = ANY(${created}::text[]) AND (intro_text LIKE ${OWN_LIKE} OR (original_invoice_id = ANY(${created}::text[]) AND EXISTS (SELECT 1 FROM invoices o WHERE o.id = invoices.original_invoice_id AND o.intro_text LIKE ${OWN_LIKE})))`; break
    case 'sentAt':     n = await prisma.$executeRaw`UPDATE invoices SET sent_at = ${patch.value} WHERE id = ${id} AND id = ANY(${created}::text[]) AND (intro_text LIKE ${OWN_LIKE} OR (original_invoice_id = ANY(${created}::text[]) AND EXISTS (SELECT 1 FROM invoices o WHERE o.id = invoices.original_invoice_id AND o.intro_text LIKE ${OWN_LIKE})))`; break
    case 'sentAtFromAudit': {
      // Nur der konkrete Versand-Audit DIESER eigenen Rechnung (FINALIZED → SENT, Benutzer-Akteur) darf die Zeitbasis liefern.
      // Das Audit-Prädikat steht zusätzlich als EXISTS im WHERE: Passt Audit und Rechnung nicht zusammen, wird KEINE Zeile
      // geändert (n = 0 → Abbruch) – sent_at wird also nie versehentlich auf NULL gesetzt.
      if (typeof patch.auditId !== 'string' || !patch.auditId) throw new Error('OWN-GUARD: sentAtFromAudit ohne auditId')
      if (!Number.isInteger(patch.deltaMs) || Math.abs(patch.deltaMs) > 3_600_000) throw new Error('OWN-GUARD: sentAtFromAudit: deltaMs ungültig')
      n = await prisma.$executeRaw`UPDATE invoices SET sent_at = (
          SELECT a.created_at - (${patch.deltaMs}::float8 * interval '1 millisecond') FROM audit_logs a
          WHERE a.id = ${patch.auditId} AND a.entity_id = ${id} AND a.entity_type = 'invoice' AND a.action = 'STATUS_CHANGE'::"AuditAction"
            AND a.user_id IS NOT NULL AND a.old_value->>'status' = 'FINALIZED' AND a.new_value->>'status' = 'SENT')
        WHERE id = ${id} AND id = ANY(${created}::text[]) AND (intro_text LIKE ${OWN_LIKE} OR (original_invoice_id = ANY(${created}::text[]) AND EXISTS (SELECT 1 FROM invoices o WHERE o.id = invoices.original_invoice_id AND o.intro_text LIKE ${OWN_LIKE})))
          AND EXISTS (SELECT 1 FROM audit_logs a2 WHERE a2.id = ${patch.auditId} AND a2.entity_id = ${id} AND a2.entity_type = 'invoice'
            AND a2.action = 'STATUS_CHANGE'::"AuditAction" AND a2.user_id IS NOT NULL
            AND a2.old_value->>'status' = 'FINALIZED' AND a2.new_value->>'status' = 'SENT')`
      break
    }
    case 'type':       n = await prisma.$executeRaw`UPDATE invoices SET type = ${patch.value}::"InvoiceType" WHERE id = ${id} AND id = ANY(${created}::text[]) AND (intro_text LIKE ${OWN_LIKE} OR (original_invoice_id = ANY(${created}::text[]) AND EXISTS (SELECT 1 FROM invoices o WHERE o.id = invoices.original_invoice_id AND o.intro_text LIKE ${OWN_LIKE})))`; break
    case 'status':     n = await prisma.$executeRaw`UPDATE invoices SET status = ${patch.value}::"InvoiceStatus" WHERE id = ${id} AND id = ANY(${created}::text[]) AND (intro_text LIKE ${OWN_LIKE} OR (original_invoice_id = ANY(${created}::text[]) AND EXISTS (SELECT 1 FROM invoices o WHERE o.id = invoices.original_invoice_id AND o.intro_text LIKE ${OWN_LIKE})))`; break
    case 'paidAmount': n = await prisma.$executeRaw`UPDATE invoices SET paid_amount = ${patch.value}::numeric WHERE id = ${id} AND id = ANY(${created}::text[]) AND (intro_text LIKE ${OWN_LIKE} OR (original_invoice_id = ANY(${created}::text[]) AND EXISTS (SELECT 1 FROM invoices o WHERE o.id = invoices.original_invoice_id AND o.intro_text LIKE ${OWN_LIKE})))`; break
    default: throw new Error('OWN-GUARD: unbekannte Operation')
  }
  if (n !== 1) throw new Error(`OWN-GUARD: ownSet(${(patch as any).op}) betraf ${n} Zeilen statt 1`)
}

// Nimmt Stornorechnungen (original_invoice_id = eigene Rechnung) in die Own-ID-Menge auf – nur wenn das Original eigen ist.
async function adoptStornos(id: string) {
  if (!created.includes(id)) return
  for (const r of await stornosOf(id)) if (!created.includes(r.id)) created.push(r.id)
}
// Storno über den echten Service; Folgerechnungen werden SOFORT registriert, auch wenn der Service oder ein späterer Schritt scheitert.
async function cancelOwn(id: string, reason: string): Promise<string[]> {
  await assertOwn(id, 'cancelOwn')
  try { await svc.cancelInvoice(id, reason, userId, userEmail) }
  finally { await adoptStornos(id) }
  return (await stornosOf(id)).map((r: any) => r.id as string)
}
// Parken: nur Rechnungen, die den Eigentumsnachweis erfüllen; Abweichung = Fehler (nach dem Versuch, alles Eigene zu parken).
async function park() {
  if (!created.length) return
  for (const id of [...created]) await adoptStornos(id)
  const n = await prisma.$executeRaw`UPDATE invoices SET due_date = ${PARK} WHERE id = ANY(${created}::text[])
    AND (intro_text LIKE ${OWN_LIKE} OR (original_invoice_id = ANY(${created}::text[]) AND EXISTS (SELECT 1 FROM invoices o WHERE o.id = invoices.original_invoice_id AND o.intro_text LIKE ${OWN_LIKE})))`
  if (n !== created.length) throw new Error(`OWN-GUARD/PARK: ${n} von ${created.length} Own-IDs geparkt – Abweichung, Befund prüfen`)
}

// ── Verschränkung (Row-Lock + pg_stat_activity) ────────────────────────────────────────────────────────────────
function holdLock(ids: string[]) {
  let release!: () => void
  const latch = new Promise<void>((r) => { release = r })
  let ready!: () => void, fail!: (e: any) => void
  let holderPid = 0
  const locked = new Promise<void>((res, rej) => { ready = res; fail = rej })
  const done = prisma.$transaction(async (tx: any) => {
    try {
      await tx.$queryRaw`SELECT id FROM invoices WHERE id = ANY(${ids}::text[]) ORDER BY id FOR UPDATE`
      holderPid = (await tx.$queryRaw`SELECT pg_backend_pid()::int AS pid`)[0].pid
      ready()
    } catch (e) { fail(e); throw e }
    await latch
  }, { timeout: 60000, maxWait: 10000 })
  return { locked, release, done, pid: () => holderPid }
}
async function lockWaiters(): Promise<Array<{ pid: number; query: string; blockers: number[] }>> {
  return prisma.$queryRaw`SELECT pid::int AS pid, query, pg_blocking_pids(pid) AS blockers FROM pg_stat_activity
    WHERE datname = current_database() AND wait_event_type = 'Lock' AND pid <> pg_backend_pid() ORDER BY pid`
}
// Poll (kein festes Schlafen als Beweis): wartet, bis genau n Sitzungen auf eine Sperre warten; sonst Abbruch mit Diagnose.
async function waitForLockWaiters(n: number) {
  const t0 = Date.now()
  for (;;) {
    const rows = await lockWaiters()
    if (rows.length === n) return rows
    if (rows.length > n || Date.now() - t0 > 15000) {
      throw new Error(`erwartet ${n} auf Sperre wartende Sitzung(en), beobachtet ${rows.length}: ${JSON.stringify(rows)}`)
    }
    await sleep(25)
  }
}
type Settled = { ok: true; value: any } | { ok: false; error: any }
function settle(p: Promise<any>) {
  const st = { finished: false }
  const res: Promise<Settled> = p.then(
    (value) => { st.finished = true; return { ok: true, value } as Settled },
    (error) => { st.finished = true; return { ok: false, error } as Settled },
  )
  return { st, res }
}
// Aktion A wird nachweislich zuerst in die Sperrwarteschlange gestellt, danach erst B; erst dann wird freigegeben.
async function orchestrate(id: string, a: () => Promise<any>, b: () => Promise<any>) {
  const lock = holdLock([id]); await lock.locked
  const holder = lock.pid()
  expect(holder).toBeGreaterThan(0)
  let A: ReturnType<typeof settle> | undefined, B: ReturnType<typeof settle> | undefined
  try {
    A = settle(a())
    const w1 = await waitForLockWaiters(1)
    const pidA = w1[0].pid
    expect(w1[0].blockers).toContain(holder)               // A wartet auf den gehaltenen Row-Lock
    expect(w1[0].query).toMatch(LOCK_QUERY)
    expect(A.st.finished).toBe(false)
    B = settle(b())                                         // erst jetzt B
    const w2 = await waitForLockWaiters(2)
    const rowB = w2.find((r) => r.pid !== pidA)!
    expect(rowB).toBeTruthy()
    expect(rowB.blockers.some((x) => x === holder || x === pidA)).toBe(true)   // B steht hinter Halter bzw. A
    expect(rowB.query).toMatch(LOCK_QUERY)
    expect(A.st.finished).toBe(false); expect(B.st.finished).toBe(false)
    lock.release(); await lock.done
    const [ra, rb] = await Promise.all([A.res, B.res])
    return { ra, rb, holder, pidA, pidB: rowB.pid }
  } finally {
    lock.release(); await lock.done.catch(() => {})
    await Promise.allSettled([A?.res, B?.res].filter(Boolean) as Promise<unknown>[])
  }
}

  // ── C9 Toleranz des Versand-Audits: reale Differenz messen, Grenzprüfung ──────────────────────────────────────
  // Gemessen wird NUR; die festgelegte Konstante (60 000 ms) wird nicht kalibriert oder verändert.
  test('C9a reale Statuswechsel FINALIZED → SENT: Differenz sentAt ↔ Audit.createdAt liegt weit innerhalb der festen Toleranz', async () => {
    const ids: string[] = []
    for (let i = 0; i < 8; i++) ids.push(await mkSent())
    const rows = await prisma.$queryRaw`SELECT i.id AS id, (EXTRACT(EPOCH FROM (a.created_at - i.sent_at)) * 1000)::float8 AS delta_ms
      FROM invoices i JOIN audit_logs a ON a.entity_id = i.id AND a.entity_type = 'invoice' AND a.action = 'STATUS_CHANGE'::"AuditAction"
        AND a.old_value->>'status' = 'FINALIZED' AND a.new_value->>'status' = 'SENT' AND a.user_id IS NOT NULL
      WHERE i.id = ANY(${ids}::text[])`
    expect(rows.length).toBe(ids.length)                                      // genau ein Nachweis je Rechnung
    const deltas = rows.map((r: any) => Number(r.delta_ms))
    for (const d of deltas) {
      expect(Math.abs(d)).toBeLessThanOrEqual(TOLERANCE_MS)                    // innerhalb der Toleranz
    }
    // Und die echte Policy akzeptiert diese regulären Nachweise tatsächlich.
    for (const id of ids.slice(0, 2)) { expect((await elig(id)).ok).toBe(true) }
    log.push(`C9a ok: ${deltas.length} reale Wechsel, Differenz Audit−sentAt: min ${Math.min(...deltas)} ms, max ${Math.max(...deltas)} ms (fachlicher Assert: ≤ ${TOLERANCE_MS} ms; diagnostischer Richtwert ${C9_DIAG_MS} ms ${Math.max(...deltas.map(Math.abs)) > C9_DIAG_MS ? 'ÜBERSCHRITTEN (nur Hinweis, kein Fehler)' : 'eingehalten'}) – Konstante unverändert`)
  })
  const EDGES: Array<[string, number, boolean]> = [
    ['+60000 ms (Grenze, inklusive)', 60_000, true],
    ['−60000 ms (Grenze, inklusive)', -60_000, true],
    ['+60001 ms', 60_001, false],
    ['−60001 ms', -60_001, false],
  ]
  for (const [label, delta, allowed] of EDGES) {
    test(`C9b Grenzprüfung gegen die echte DB: sentAt = Audit.createdAt ${label} → ${allowed ? 'mahnbar' : 'NO_SEND_AUDIT'}`, async () => {
      const id = await mkSent()
      const ev = await dispatchEvidence(id); expect(ev.length).toBe(1)
      // Differenz Audit − sentAt = delta; sent_at wird direkt in PostgreSQL aus dem konkreten Audit-Zeitstempel berechnet (nur eigene Testzeile)
      await ownSet(id, { op: 'sentAtFromAudit', auditId: ev[0].audit_id, deltaMs: delta })
      const d = (await prisma.$queryRaw`SELECT (EXTRACT(EPOCH FROM (a.created_at - i.sent_at)) * 1000)::float8 AS d FROM invoices i
        JOIN audit_logs a ON a.id = ${ev[0].audit_id} WHERE i.id = ${id}`)[0].d
      expect(Number(d)).toBe(delta)
      if (allowed) {
        const nid = await mkDun(id, 1)
        expect((await createAudits(nid)).length).toBe(1)
        expect((await elig(id)).ok).toBe(false)                                       // nach Anlage: nächste Stufe wartet auf „versendet“
      } else {
        const s0 = await stateOf(id)
        const e = await rejection(mkDun(id, 1))
        expect(e.name).toBe('BusinessRuleError'); expect(e.message).toMatch(/nicht nachvollziehbar dokumentiert/)
        expect(await stateOf(id)).toBe(s0)
        expect((await elig(id)).reasons).toContain('NO_SEND_AUDIT')
      }
      log.push(`C9b ok: ${label} → ${allowed ? 'mahnbar' : 'gesperrt'}`)
    })
  }

  })
